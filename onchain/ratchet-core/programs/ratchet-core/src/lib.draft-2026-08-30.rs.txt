pub const MAX_PESSIMISTIC_XP_PENALTY: u64 = 50_000_000; // Scaled by 1M
pub const PESSIMISTIC_BRIER_DENOMINATOR_PENALTY: u32 = 1;

#[program]

    /// G2-02: Accept Atomically
    /// Binds owner, target, stake, sealed probability/side, and Pyth oracle entry.
    pub fn accept_shot(
        ctx: Context<AcceptShot>,
        command_nonce: u64,
        stake: u64,
        feed_id: [u8; 32],
        entry_price: u64,
        expiry_ts: i64,
        commit_hash: [u8; 32],
    ) -> Result<()> {
        let clock = Clock::get()?;
        require!(expiry_ts > clock.unix_timestamp, CoreError::ExpiryInPast);
        require!(stake > 0, CoreError::InvalidStake);

        // Deduct stake from player's token account into the global vault
        let cpi_accounts = Transfer {
            from: ctx.accounts.player_token_account.to_account_info(),
            to: ctx.accounts.vault_token_account.to_account_info(),
            authority: ctx.accounts.player.to_account_info(),
        };
        let cpi_program = ctx.accounts.token_program.to_account_info();
        token::transfer(CpiContext::new(cpi_program, cpi_accounts), stake)?;

        let shot = &mut ctx.accounts.shot;
        shot.player = ctx.accounts.player.key();
        shot.command_nonce = command_nonce;
        shot.stake = stake;
        shot.feed_id = feed_id;
        shot.entry_price = entry_price;
        shot.expiry_ts = expiry_ts;
        shot.commit_hash = commit_hash;
        shot.status = ShotStatus::Open as u8;
        shot.checkpointed_price = 0;

        // Apply Pessimistic Deduction to PlayerLedger to penalize non-reveals (Q03)
        let ledger = &mut ctx.accounts.player_ledger;
        if ledger.player == Pubkey::default() {
            ledger.player = ctx.accounts.player.key();
        }
        ledger.xp = ledger.xp.saturating_sub(MAX_PESSIMISTIC_XP_PENALTY);
        ledger.brier_denominator = ledger.brier_denominator.saturating_add(PESSIMISTIC_BRIER_DENOMINATOR_PENALTY);

        Ok(())
    }

    /// G2-03: Checkpoint Race & Settle/VOID Atomically
    /// The Pyth Push oracle `FeedClock` logic provides the checkpointed price.
    /// This function handles the economic settlement and reveal.
    pub fn settle_shot(
        ctx: Context<SettleShot>,
        side: u8,
        probability_e5: u32,
        salt: [u8; 32],
        checkpointed_price: u64, // Assume validated by CPI or feed clock in real implementation
    ) -> Result<()> {
        let shot = &mut ctx.accounts.shot;
        require!(shot.status == ShotStatus::Open as u8, CoreError::InvalidState);
        
        let side_bytes: &[u8] = if side == 1 { b"YES" } else { b"NO" };
        let prob_str = probability_e5.to_string();
        
        // Verify Reveal
        let commitment = hashv(&[
            b"RATCHET|v3|",
            shot.player.as_ref(),
            b"|",
            &shot.command_nonce.to_le_bytes(),
            b"|",
            side_bytes,
            b"|",
            prob_str.as_bytes(),
            b"|",
            &salt,
        ]);
        require!(commitment.to_bytes() == shot.commit_hash, CoreError::CommitMismatch);

        shot.checkpointed_price = checkpointed_price;
        
        // G2-03 & Q04: Equality == VOID
        if checkpointed_price == shot.entry_price {
            return void_internal_context(ctx.accounts.player_token_account.clone(), ctx.accounts.vault_token_account.clone(), ctx.accounts.token_program.clone(), ctx.accounts.player_ledger.clone(), shot, ctx.bumps.vault_token_account);
        }

        let outcome_yes = checkpointed_price > shot.entry_price;
        let hit = (side == 1) == outcome_yes;

        shot.status = ShotStatus::Settled as u8;

        // Calculate actual XP and payout (simplified for testing)
        // Reverses the pessimistic deduction and applies the actual score.
        let actual_xp = if hit { 1_000_000 } else { 0 }; // Example logic
        
        let ledger = &mut ctx.accounts.player_ledger;
        ledger.xp = ledger.xp.saturating_add(MAX_PESSIMISTIC_XP_PENALTY).saturating_add(actual_xp);
        ledger.shots = ledger.shots.saturating_add(1);
        if hit {
            ledger.hits = ledger.hits.saturating_add(1);
            let payout = shot.stake.checked_mul(170).unwrap().checked_div(100).unwrap();
            
            // Transfer winnings back
            let seeds = &[b"vault".as_ref(), &[ctx.bumps.vault_token_account]];
            let signer = &[&seeds[..]];
            let cpi_accounts = Transfer {
                from: ctx.accounts.vault_token_account.to_account_info(),
                to: ctx.accounts.player_token_account.to_account_info(),
                authority: ctx.accounts.vault_token_account.to_account_info(), // Vault PDA
            };
            let cpi_program = ctx.accounts.token_program.to_account_info();
            token::transfer(CpiContext::new_with_signer(cpi_program, cpi_accounts, signer), payout)?;
        }

        Ok(())
    }

    pub fn grant_delegate(ctx: Context<GrantDelegate>, allowance: u64, max_per_shot: u64, expiry_ts: i64, methods: u8) -> Result<()> {
        let grant = &mut ctx.accounts.grant;

        grant.allowance_remaining = allowance;
        grant.max_stake_per_shot = max_per_shot;
        grant.expiry_ts = expiry_ts;
        grant.methods = methods;
        Ok(())
    }

    /// G2-03: Settle/VOID Atomically (Void Path)

    pub fn initialize_migration(ctx: Context<InitializeMigration>, root: [u8; 32]) -> Result<()> {
        let state = &mut ctx.accounts.migration_state;
        state.authority = ctx.accounts.authority.key();
        state.root = root;
        Ok(())
    }

    pub fn update_migration_root(ctx: Context<UpdateMigrationRoot>, root: [u8; 32]) -> Result<()> {
        require!(ctx.accounts.migration_state.authority == ctx.accounts.authority.key(), CoreError::Unauthorized);
        ctx.accounts.migration_state.root = root;
        Ok(())
    }

    pub fn claim_legacy_balance(ctx: Context<ClaimLegacyBalance>, cr: u64, xp: u64, proof: Vec<[u8; 32]>) -> Result<()> {
        let mut leaf_data = [0u8; 48];
        leaf_data[0..32].copy_from_slice(&ctx.accounts.player.key().to_bytes());
        leaf_data[32..40].copy_from_slice(&cr.to_le_bytes());
        leaf_data[40..48].copy_from_slice(&xp.to_le_bytes());
        let leaf = solana_program::hash::hashv(&[&leaf_data]).to_bytes();

        require!(verify_proof(&proof, &ctx.accounts.migration_state.root, &leaf), CoreError::InvalidMerkleProof);

        let claim = &mut ctx.accounts.claim;
        require!(!claim.claimed, CoreError::AlreadyClaimed);
        claim.claimed = true;

        let ledger = &mut ctx.accounts.player_ledger;
        if ledger.player == Pubkey::default() {
            ledger.player = ctx.accounts.player.key();
        }
        ledger.xp = ledger.xp.saturating_add(xp);

        if cr > 0 {
            let seeds = &[b"vault".as_ref(), &[ctx.bumps.vault_token_account]];
            let signer = &[&seeds[..]];
            let cpi_accounts = Transfer {
                from: ctx.accounts.vault_token_account.to_account_info(),
                to: ctx.accounts.player_token_account.to_account_info(),
                authority: ctx.accounts.vault_token_account.to_account_info(),
            };
            let cpi_program = ctx.accounts.token_program.to_account_info();
            token::transfer(CpiContext::new_with_signer(cpi_program, cpi_accounts, signer), cr)?;
        }

        Ok(())
    }

    pub fn void_shot(ctx: Context<VoidShot>) -> Result<()> {
        let clock = Clock::get()?;
        let shot = &mut ctx.accounts.shot;
        require!(shot.status == ShotStatus::Open as u8, CoreError::InvalidState);
        require!(clock.unix_timestamp >= shot.expiry_ts + 900, CoreError::NotVoidableYet);

        void_internal_context(ctx.accounts.player_token_account.clone(), ctx.accounts.vault_token_account.clone(), ctx.accounts.token_program.clone(), ctx.accounts.player_ledger.clone(), shot, ctx.bumps.vault_token_account)
    }
}


fn verify_proof(proof: &[[u8; 32]], root: &[u8; 32], leaf: &[u8; 32]) -> bool {
    let mut computed_hash = *leaf;
    for proof_element in proof.iter() {
        if computed_hash <= *proof_element {
            computed_hash = solana_program::hash::hashv(&[&computed_hash, proof_element]).to_bytes();
        } else {
            computed_hash = solana_program::hash::hashv(&[proof_element, &computed_hash]).to_bytes();
        }
    }
    computed_hash == *root
}

// Helper to execute VOID (Refund and reverse penalty)
fn void_internal<'a>(ctx: Context<SettleShot<'a>>, shot: &mut Account<'a, Shot>) -> Result<()> {
    void_internal_context(ctx.accounts.player_token_account.clone(), ctx.accounts.vault_token_account.clone(), ctx.accounts.token_program.clone(), ctx.accounts.player_ledger.clone(), shot, ctx.bumps.vault_token_account)
}

fn void_internal_context<'info>(
    player_token_account: Account<'info, TokenAccount>,
    vault_token_account: Account<'info, TokenAccount>,
    token_program: Program<'info, Token>,
    mut player_ledger: Account<'info, PlayerLedger>,
    shot: &mut Account<'info, Shot>,
    vault_bump: u8,
) -> Result<()> {
    shot.status = ShotStatus::Voided as u8;

    // Refund stake
    let seeds = &[b"vault".as_ref(), &[vault_bump]];
    let signer = &[&seeds[..]];
    let cpi_accounts = Transfer {
        from: vault_token_account.to_account_info(),
        to: player_token_account.to_account_info(),
        authority: vault_token_account.to_account_info(),
    };
    let cpi_program = token_program.to_account_info();
    token::transfer(CpiContext::new_with_signer(cpi_program, cpi_accounts, signer), shot.stake)?;

    // Reverse pessimistic deduction completely
    player_ledger.xp = player_ledger.xp.saturating_add(MAX_PESSIMISTIC_XP_PENALTY);
    player_ledger.brier_denominator = player_ledger.brier_denominator.saturating_sub(PESSIMISTIC_BRIER_DENOMINATOR_PENALTY);

    Ok(())
}
